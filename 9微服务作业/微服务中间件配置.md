
使用 Docker Compose 来统一管理这些高频中间件是非常棒的做法，一键启动，互不干扰，而且它们之间可以通过内部网络直接相互通信。

下面为你量身定制的 `docker-compose.yml` 文件。我已经为你配置好了持久化目录（防止数据丢失）、时区（Asia/Shanghai），并将你要求的密码 `zxcvbnmko` 写入了配置文件中。

### 1. 创建文件

在你的服务器或本地电脑上新建一个空文件夹，在里面创建一个名为 `docker-compose.yml` 的文件，将以下内容复制进去：



``` yaml
#version: "3.8"

  

# 定义一个通用的日志配置模板

x-logging: &default-logging

  driver: "json-file"

  options:

    max-size: "10m"

    max-file: "3"

  

services:

  # 1. MySQL 8.0 服务

  mysql:

    image: mysql:8.0

    container_name: dev-mysql

    logging: *default-logging

    environment:

      MYSQL_ROOT_PASSWORD: ${DEV_PASSWORD} # <-- 动态引入变量

      TZ: Asia/Shanghai

    ports:

      - "3307:3306"

    volumes:

      - ./data/mysql:/var/lib/mysql

      - ./mysql-init:/docker-entrypoint-initdb.d

    networks:

      - dev_network

    # --- 健康检查命令也同步支持变量密码 ---

    healthcheck:

      test:

        [

          "CMD",

          "mysqladmin",

          "ping",

          "-h",

          "localhost",

          "-u",

          "root",

          "-p${DEV_PASSWORD}",

        ]

      interval: 5s

      timeout: 5s

      retries: 5

  

  # 2. Redis 服务

  redis:

    image: redis:7-alpine

    container_name: dev-redis

    logging: *default-logging

    command: redis-server --requirepass ${DEV_PASSWORD} # <-- 动态引入变量

    ports:

      - "6379:6379"

    volumes:

      - ./data/redis:/data

    networks:

      - dev_network

  

  # 3. RabbitMQ 服务

  rabbitmq:

    image: rabbitmq:3-management

    container_name: dev-rabbitmq

    logging: *default-logging

    environment:

      RABBITMQ_DEFAULT_USER: admin

      RABBITMQ_DEFAULT_PASS: ${DEV_PASSWORD} # <-- 动态引入变量

      TZ: Asia/Shanghai

  

    ports:

      - "5672:5672"

      - "15672:15672"

    volumes:

      - ./data/rabbitmq:/var/lib/rabbitmq

    networks:

      - dev_network

  

  # 4. Nacos 服务

  nacos:

    image: nacos/nacos-server:v2.3.2

    container_name: dev-nacos

    logging: *default-logging

    environment:

      MODE: standalone

      TZ: Asia/Shanghai

      JVM_XMS: 256m

      JVM_XMX: 256m

      NACOS_AUTH_ENABLE: "true"

      NACOS_AUTH_TOKEN: ${NACOS_TOKEN} # <-- 动态引入安全密钥

      NACOS_AUTH_IDENTITY_KEY: "nacos"

      NACOS_AUTH_IDENTITY_VALUE: "nacos"

    ports:

      - "8848:8848"

      - "9848:9848"

      - "9849:9849"

    volumes:

      - ./data/nacos:/home/nacos/data

    networks:

      - dev_network

  

  # 5. XXL-JOB 任务调度中心

  xxl-job-admin:

    image: xuxueli/xxl-job-admin:2.4.1

    container_name: dev-xxl-job

    logging: *default-logging

    ports:

      - "9090:8080"

    environment:

      # --- 多行字符串中同样完美支持变量解析 ---

      PARAMS: "

        --spring.datasource.url=jdbc:mysql://dev-mysql:3306/xxl_job?useUnicode=true&characterEncoding=UTF-8&autoReconnect=true&serverTimezone=Asia/Shanghai

        --spring.datasource.username=root

        --spring.datasource.password=${DEV_PASSWORD}

        "

    networks:

      - dev_network

    depends_on:

      mysql:

        condition: service_healthy

  

  # 6. MinIO 对象存储服务

  minio:

    image: minio/minio

    container_name: dev-minio

    logging: *default-logging

    environment:

      MINIO_ROOT_USER: admin # MinIO 登录账号

      MINIO_ROOT_PASSWORD: ${DEV_PASSWORD} # <-- 动态引入变量作为登录密码

      TZ: Asia/Shanghai

    # 启动命令：指定数据存放目录，并固定 Web 控制台的端口

    command: server /data --console-address ":9001"

    ports:

      - "9000:9000" # API 通讯端口（代码里用的就是这个）

      - "9001:9001" # Web 浏览器控制台访问端口

    volumes:

      - ./data/minio:/data # 数据持久化目录

    networks:

      - dev_network

  

networks:

  dev_network:

    driver: bridge
```

.env 文件

```
# ==========================================
# 微服务基础环境统一变量配置
# ==========================================

# 统一通用密码 (应用于 MySQL、Redis、RabbitMQ)
DEV_PASSWORD=zxcvbnmko

# Nacos 核心安全 Token (必须大于 32 位字符串)
NACOS_TOKEN=SecretKey012345678901234567890123456789012345678901234567890123456789
```

| -------------------------------------- | ------------------------------------- | -------- | ----------------- | ------------------------------------------------------- |
| **MySQL (8.0)**                        | `127.0.0.1:3307`                      | `root`   | `${DEV_PASSWORD}` | 关系型数据库。注意：宿主机映射端口为 **3307**，避免了与本地默认的 3306 冲突。          |
| **Redis (7)**                          | `127.0.0.1:6379`                      | _(无)_    | `${DEV_PASSWORD}` | 缓存服务。连接时需提供配置的密码进行 AUTH 认证。                             |
| **RabbitMQ**<br><br>  <br><br>_(管理台)_  | `http://localhost:15672`              | `admin`  | `${DEV_PASSWORD}` | 消息队列 Web 管理控制台。                                         |
| **RabbitMQ**<br><br>  <br><br>_(应用连接)_ | `127.0.0.1:5672`                      | `admin`  | `${DEV_PASSWORD}` | 微服务代码中配置的 AMQP 消息队列通讯端口。                                |
| **Nacos (2.3.2)**                      | `http://localhost:8848/nacos`         | `nacos`  | `nacos` _(系统默认)_  | 注册中心与配置中心 Web 控制台。内部 API 通讯需携带 Token: `${NACOS_TOKEN}`。 |
| **XXL-JOB**                            | `http://localhost:9090/xxl-job-admin` | `admin`  | `123456` _(系统默认)_ | 分布式任务调度中心。_(注：需确保 MySQL 中已提前初始化 `xxl_job` 库)_           |
| **MinIO**<br><br>  <br><br>_(管理台)_     | `http://localhost:9001`               | `admin`  | `${DEV_PASSWORD}` | 对象存储 Web 管理控制台。用于手动管理 Bucket 和文件。                       |
| **MinIO**<br><br>  <br><br>_(应用连接)_    | `127.0.0.1:9000`                      | `admin`  | `${DEV_PASSWORD}` | 微服务代码中（如通过 SDK）配置文件上传/下载的 API 端口。                       |
