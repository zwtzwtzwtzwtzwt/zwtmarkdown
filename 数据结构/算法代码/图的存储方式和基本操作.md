这是一份为您整合并重新排版的图存储结构“定义 + 增删改查操作”完整复习指南。内容严格遵循**严蔚敏《数据结构》**和**王道考研**的 C 语言标准，涵盖了数据结构考研（如 408 及自主命题）中最核心的手写代码考点。

图的操作中，省略了内存分配失败的严谨校验（考试默认分配成功），核心聚焦于**逻辑和指针的变换**，插入链表结点默认采用**头插法**。

## 1. 邻接矩阵法 (Adjacency Matrix)

**适用场景与考点：** 适合**稠密图**。无向图的邻接矩阵是对称的，考研选择题常考对称矩阵的压缩存储计算；空间复杂度为 $O(|V|^2)$，与边数无关。

> **避坑指南：** 在考卷上手写图的算法时，如果题目没有说明存储结构，**优先选用邻接矩阵**。它的下标操作最直观，且判别两个顶点之间是否有边的时间复杂度为 $O(1)$，极大地降低了手写代码的出错率。考研极少要求基于邻接矩阵手写删除顶点代码（时间复杂度高达 $O(|V|)$，需大面积移动元素）。

### 1.1 结构体定义

C

```
typedef struct {
    VertexType Vex[MaxVertexNum];               // 顶点表 (一维数组)
    EdgeType Edge[MaxVertexNum][MaxVertexNum];  // 邻接矩阵，边表 (二维数组)
    int vexnum, arcnum;                         // 图的当前顶点数和弧数
} MGraph;
```

### 1.2 核心操作代码 (以无向图为例)

C

```
// 1. 初始化 (创)
void InitGraph(MGraph *G) {
    G->vexnum = 0;
    G->arcnum = 0;
    for (int i = 0; i < MaxVertexNum; i++) {
        for (int j = 0; j < MaxVertexNum; j++) {
            G->Edge[i][j] = 0; // 无权图初始化为0，带权图应初始化为无穷大(INF)
        }
    }
}

// 2. 插入新顶点 (增)
bool InsertVex(MGraph *G, VertexType v) {
    if (G->vexnum >= MaxVertexNum) return false;
    G->Vex[G->vexnum++] = v;   // 将顶点放入数组末尾
    return true;
}

// 3. 插入边/弧 (增)
bool InsertEdge(MGraph *G, int x, int y) {
    if (x < 0 || x >= G->vexnum || y < 0 || y >= G->vexnum) return false;
    if (G->Edge[x][y] == 0) {
        G->Edge[x][y] = 1;
        G->Edge[y][x] = 1;     // 无向图需要对称赋值。若为有向图则删去此行
        G->arcnum++;
    }
    return true;
}

// 4. 删除边/弧 (删)
bool DeleteEdge(MGraph *G, int x, int y) {
    if (x < 0 || x >= G->vexnum || y < 0 || y >= G->vexnum) return false;
    if (G->Edge[x][y] == 1) {
        G->Edge[x][y] = 0;
        G->Edge[y][x] = 0;     // 无向图对称删除
        G->arcnum--;
    }
    return true;
}

// 5. 查找是否存在边 (查)
bool Adjacent(MGraph G, int x, int y) {
    if (x < 0 || x >= G.vexnum || y < 0 || y >= G.vexnum) return false;
    return G.Edge[x][y] == 1;
}
```

## 2. 邻接表法 (Adjacency List)

**适用场景与考点：** 适合**稀疏图**。空间复杂度为 $O(|V| + |E|)$（有向图）或 $O(|V| + 2|E|)$（无向图）。对于有向图，求结点的出度很容易，但求入度需要遍历整个邻接表。

> **避坑指南：** 邻接表是不唯一的！因为每个顶点的边表结点顺序可以任意排布。这也导致了基于邻接表的 DFS 和 BFS 遍历序列是不唯一的（考研大题常设的陷阱）。

### 2.1 结构体定义

C

```
// 边表结点
typedef struct ArcNode {
    int adjvex;                 // 该弧所指向的顶点的位置（数组下标）
    struct ArcNode *nextarc;    // 指向下一条弧的指针
    // int weight;              // 网的边权值，如果是带权图需加上此属性
} ArcNode;

// 顶点表结点
typedef struct VNode {
    VertexType data;            // 顶点信息
    ArcNode *firstarc;          // 指向第一条依附该顶点的弧的指针
} VNode, AdjList[MaxVertexNum];

// 邻接表表示的图
typedef struct {
    AdjList vertices;           // 邻接表 (即顶点数组)
    int vexnum, arcnum;         // 图的当前顶点数和弧数
} ALGraph;
```

### 2.2 核心操作代码 (以有向图为例)

C

```
// 1. 初始化 (创)
void InitALGraph(ALGraph *G) {
    G->vexnum = 0;
    G->arcnum = 0;
    for (int i = 0; i < MaxVertexNum; i++) {
        G->vertices[i].firstarc = NULL; // 初始时所有边表均为空
    }
}

// 2. 插入新顶点 (增)
bool InsertVex(ALGraph *G, VertexType v) {
    if (G->vexnum >= MaxVertexNum) return false;
    G->vertices[G->vexnum].data = v;
    G->vertices[G->vexnum].firstarc = NULL;
    G->vexnum++;
    return true;
}

// 3. 插入弧 <x, y> (增) - 考点：头插法
bool InsertArc(ALGraph *G, int x, int y) {
    ArcNode *p = (ArcNode *)malloc(sizeof(ArcNode));
    p->adjvex = y;
    // 头插法将节点插入到顶点 x 的出边表中
    p->nextarc = G->vertices[x].firstarc;
    G->vertices[x].firstarc = p;
    G->arcnum++;
    return true;
}

// 4. 删除弧 <x, y> (删) - 考点：单链表的删除
bool DeleteArc(ALGraph *G, int x, int y) {
    ArcNode *p = G->vertices[x].firstarc;
    ArcNode *pre = NULL;
    while (p != NULL && p->adjvex != y) { // 寻找终点为 y 的边表结点
        pre = p;
        p = p->nextarc;
    }
    if (p == NULL) return false;          // 没找到这条边
    
    if (pre == NULL) {                    // 要删除的是第一个结点
        G->vertices[x].firstarc = p->nextarc;
    } else {                              // 要删除的不是第一个结点
        pre->nextarc = p->nextarc;
    }
    free(p);
    G->arcnum--;
    return true;
}

// 5. 查找是否存在弧 <x, y> (查)
bool Adjacent(ALGraph G, int x, int y) {
    ArcNode *p = G.vertices[x].firstarc;
    while (p != NULL) {
        if (p->adjvex == y) return true;
        p = p->nextarc;
    }
    return false;
}
```

## 3. 十字链表 (Orthogonal List)

**适用场景与考点：** 专门用于存储**有向图**。它完美解决了邻接表“求入度难”的问题。在十字链表中，既容易找到以 $V_i$ 为尾的弧（出度），也容易找到以 $V_i$ 为头的弧（入度）。

> **记忆技巧：** 弧结点的 5 个域可以记为：“**尾、头、同头、同尾、(信息)**”。
> 
> **考场避坑：** 考研绝不会让你手写十字链表的“删除顶点”，因为牵扯的入出度指针太多。最常考的是“**插入弧**”操作（如何同时修改出边表和入边表）。

### 3.1 结构体定义

C

```
// 弧结点 (边)
typedef struct ArcNode {
    int tailvex, headvex;         // 该弧的尾(起点)和头(终点)的顶点位置
    struct ArcNode *hlink, *tlink;// 分别指向弧头相同的下一条弧、弧尾相同的下一条弧
    // int weight;                // 权值
} ArcNode;

// 顶点结点
typedef struct VNode {
    VertexType data;              // 顶点信息
    ArcNode *firstin, *firstout;  // 分别指向以该顶点为头(入)和尾(出)的第一条弧
} VNode;

// 十字链表表示的有向图
typedef struct {
    VNode xlist[MaxVertexNum];    // 顶点数组
    int vexnum, arcnum;           // 图的当前顶点数和弧数
} GLGraph;
```

### 3.2 核心操作代码

C

```
// 1. 插入弧 <x, y> (核心考点)
bool InsertArc_GL(GLGraph *G, int x, int y) {
    ArcNode *p = (ArcNode *)malloc(sizeof(ArcNode));
    p->tailvex = x;    // 弧尾
    p->headvex = y;    // 弧头
    
    // ① 将该弧用头插法插入到 x 的【出边表】中
    p->tlink = G->xlist[x].firstout;
    G->xlist[x].firstout = p;
    
    // ② 将该弧用头插法插入到 y 的【入边表】中
    p->hlink = G->xlist[y].firstin;
    G->xlist[y].firstin = p;
    
    G->arcnum++;
    return true;
}

// 2. 查找是否存在弧 <x, y>
bool Adjacent_GL(GLGraph G, int x, int y) {
    // 既可以查 x 的出度表，也可以查 y 的入度表，这里查 x 的出度
    ArcNode *p = G.xlist[x].firstout;
    while (p != NULL) {
        if (p->headvex == y) return true;
        p = p->tlink; // 沿着同弧尾的链表往下走
    }
    return false;
}
```

## 4. 邻接多重表 (Adjacency Multilist)

**适用场景与考点：** 专门用于存储**无向图**。在普通的邻接表中，无向图的一条边会被存储两次，删除或修改边非常麻烦。邻接多重表让每条边只存在一个结点，彻底解决了这个问题。

> **记忆技巧：** 边结点的域可以对称着记：“**标记、节点 i、节点 j、连着 i 的边、连着 j 的边**”。
> 
> **操作重点：** 与十字链表类似，重点在于“**插入无向边**”时，如何把这一个结点同时链入两个顶点的依附边链表中。

### 4.1 结构体定义

C

```
// 边结点
typedef struct EBox {
    int mark;                     // 标志域，可用以标记该边是否被搜索过
    int ivex, jvex;               // 该边依附的两个顶点在数组中的位置
    struct EBox *ilink, *jlink;   // 分别指向依附于 ivex 和 jvex 的下一条边
    // int weight;                // 权值
} EBox;

// 顶点结点
typedef struct VexBox {
    VertexType data;              // 顶点信息
    EBox *firstedge;              // 指向第一条依附该顶点的边
} VexBox;

// 邻接多重表表示的无向图
typedef struct {
    VexBox adjmulist[MaxVertexNum]; // 顶点数组
    int vexnum, arcnum;             // 图的当前顶点数和边数
} AMLGraph;
```

### 4.2 核心操作代码

C

```
// 1. 插入无向边 (x, y) (核心考点)
bool InsertEdge_AML(AMLGraph *G, int x, int y) {
    EBox *p = (EBox *)malloc(sizeof(EBox));
    p->mark = 0;       // 初始化未被访问
    p->ivex = x;       // 依附的端点之一
    p->jvex = y;       // 依附的另一个端点
    
    // ① 将该边用头插法插入到顶点 x 的链表中
    p->ilink = G->adjmulist[x].firstedge;
    G->adjmulist[x].firstedge = p;
    
    // ② 将该边用头插法插入到顶点 y 的链表中
    p->jlink = G->adjmulist[y].firstedge;
    G->adjmulist[y].firstedge = p;
    
    G->arcnum++;
    return true;
}

// 2. 查找是否存在边 (x, y)
bool Adjacent_AML(AMLGraph G, int x, int y) {
    EBox *p = G.adjmulist[x].firstedge;
    while (p != NULL) {
        // 因为是无向边，p->ivex 可能是 x 也可能是 y
        if ((p->ivex == x && p->jvex == y) || (p->ivex == y && p->jvex == x)) {
            return true;
        }
        // 指针沿着依附于 x 的边继续往下走
        p = (p->ivex == x) ? p->ilink : p->jlink; 
    }
    return false;
}
```
