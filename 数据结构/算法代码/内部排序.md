备战计算机考研（如 408 或自主命题），掌握经典排序算法的底层逻辑和手写代码是必修课。这里的代码主要参照**王道考研**和**严蔚敏《数据结构》**的标准 C/C++ 实现风格，注重**规范性**、**边界处理**和**应试书写标准**。

为了方便复用，我们统一定义一个交换函数（部分手写卷面中可以直接调用 `swap`）：


```C++
void swap(int &a, int &b) {
    int temp = a;
    a = b;
    b = temp;
}
```

### 1. 插入排序 (Insertion Sort)

**考点提示：** 考研中常考折半插入优化，这里给出最基础的直接插入排序。默认数组下标从 `0` 开始。


```C++
void InsertSort(int A[], int n) {
    int i, j, temp;
    for (i = 1; i < n; i++) {
        if (A[i] < A[i - 1]) {        // 若A[i]关键字小于前驱
            temp = A[i];              // 暂存A[i]
            for (j = i - 1; j >= 0 && A[j] > temp; --j) {
                A[j + 1] = A[j];      // 所有大于temp的元素都向后挪位
            }
            A[j + 1] = temp;          // 复制到插入位置
        }
    }
}
```

### 2. 希尔排序 (Shell Sort)

**考点提示：** 本质是分组的插入排序。常考增量序列（通常是每次减半），不稳定排序。


```C++
void ShellSort(int A[], int n) {
    int d, i, j, temp;
    // d 为增量，初始为n/2，每次减半，直到1
    for (d = n / 2; d >= 1; d = d / 2) {
        for (i = d; i < n; i++) {
            if (A[i] < A[i - d]) {     // 需将A[i]插入有序增量子表
                temp = A[i];
                for (j = i - d; j >= 0 && A[j] > temp; j -= d) {
                    A[j + d] = A[j];   // 记录后移，查找插入位置
                }
                A[j + d] = temp;       // 插入
            }
        }
    }
}
```

### 3. 冒泡排序 (Bubble Sort)

**考点提示：** 必须加上 `flag` 标志位优化，这是考研给分点，表示如果某一趟没有发生交换，说明全局已有序，提前结束。


```C++
void BubbleSort(int A[], int n) {
    for (int i = 0; i < n - 1; i++) {
        bool flag = false;            // 表示本趟冒泡是否发生交换的标志
        for (int j = n - 1; j > i; j--) { // 一趟冒泡过程（从后往前）
            if (A[j - 1] > A[j]) {    // 若为逆序
                swap(A[j - 1], A[j]); // 交换
                flag = true;
            }
        }
        if (flag == false) return;    // 本趟遍历后没有发生交换，说明表已经有序
    }
}
```

### 4. 快速排序 (Quick Sort)

**考点提示：** 408 最爱考的代码之一！核心是 `Partition` 划分函数。教材习惯用第一个元素作为 Pivot。


```C++
// 划分操作
int Partition(int A[], int low, int high) {
    int pivot = A[low];               // 将当前表中第一个元素设为枢轴，对表进行划分
    while (low < high) {              // 循环跳出条件
        while (low < high && A[high] >= pivot) --high;
        A[low] = A[high];             // 将比枢轴小的元素移动到左端
        while (low < high && A[low] <= pivot) ++low;
        A[high] = A[low];             // 将比枢轴大的元素移动到右端
    }
    A[low] = pivot;                   // 枢轴元素存放到最终位置
    return low;                       // 返回存放枢轴的最终位置
}

void QuickSort(int A[], int low, int high) {
    if (low < high) {                 // 递归跳出的条件
        int pivotpos = Partition(A, low, high); // 划分
        QuickSort(A, low, pivotpos - 1);        // 依次对两个子表进行递归排序
        QuickSort(A, pivotpos + 1, high);
    }
}
```

### 5. 简单选择排序 (Simple Selection Sort)

**考点提示：** 每一趟在后面选一个最小的放到前面。不稳定排序。


```C++
void SelectSort(int A[], int n) {
    for (int i = 0; i < n - 1; i++) { // 一共进行n-1趟
        int min = i;                  // 记录最小元素位置
        for (int j = i + 1; j < n; j++) {
            if (A[j] < A[min]) min = j; // 更新最小元素位置
        }
        if (min != i) swap(A[i], A[min]); // 将最小元素与第i个位置交换
    }
}
```

### 6. 堆排序及堆的插入与删除 (Heap Sort)

**⚠️ 极重要提示：** 考研教材中为了方便计算左右孩子（$2i$ 与 $2i+1$），**堆排序的数组一律要求下标从 1 开始！** 即元素存在 `A[1]` 到 `A[len]`，`A[0]` 通常作为暂存单元。以下是大根堆实现。


```C++
// 1. 堆的向下调整 (核心)
void HeadAdjust(int A[], int k, int len) {
    A[0] = A[k];                      // A[0]暂存子树的根结点
    for (int i = 2 * k; i <= len; i *= 2) { // 沿key较大的子结点向下筛选
        if (i < len && A[i] < A[i + 1]) 
            i++;                      // 取key较大的子结点的下标
        if (A[0] >= A[i]) break;      // 筛选结束
        else {
            A[k] = A[i];              // 将A[i]调整到双亲结点上
            k = i;                    // 修改k值，以便继续向下筛选
        }
    }
    A[k] = A[0];                      // 被筛选结点的值放入最终位置
}

// 2. 建初堆
void BuildMaxHeap(int A[], int len) {
    for (int i = len / 2; i > 0; i--) // 从后往前调整所有非叶结点
        HeadAdjust(A, i, len);
}

// 3. 堆排序
void HeapSort(int A[], int len) {
    BuildMaxHeap(A, len);             // 初始建堆
    for (int i = len; i > 1; i--) {   // n-1趟的交换和建堆过程
        swap(A[i], A[1]);             // 堆顶元素和堆底元素交换
        HeadAdjust(A, 1, i - 1);      // 把剩余的i-1个元素整理成堆
    }
}

// 4. 堆的插入 (向上调整)
void HeapInsert(int A[], int &len, int val) {
    A[++len] = val;                   // 先放到数组末尾，堆长度+1
    int i = len;
    while (i > 1 && A[i / 2] < A[i]) { // 若比父节点大，则不断往上浮
        swap(A[i / 2], A[i]);
        i = i / 2;
    }
}

// 5. 堆的删除 (通常指删除堆顶)
int HeapDelete(int A[], int &len) {
    int top = A[1];                   // 记录堆顶
    A[1] = A[len--];                  // 将末尾元素移到堆顶，堆长度-1
    HeadAdjust(A, 1, len);            // 向下调整恢复堆
    return top;
}
```

### 7. 归并排序 (Merge Sort)

**考点提示：** 必须使用辅助数组 `B`，空间复杂度为 $O(n)$。归并也是外排序的核心算法。


```C++
int *B = (int *)malloc((n + 1) * sizeof(int)); // 辅助数组B

void Merge(int A[], int low, int mid, int high) {
    int i, j, k;
    for (k = low; k <= high; k++) B[k] = A[k]; // 将A中所有元素复制到B中
    
    for (i = low, j = mid + 1, k = i; i <= mid && j <= high; k++) {
        if (B[i] <= B[j]) A[k] = B[i++];       // 比较B的左右两段中的元素，较小值复制回A
        else A[k] = B[j++];
    }
    while (i <= mid)  A[k++] = B[i++];         // 若第一个表未检测完，复制
    while (j <= high) A[k++] = B[j++];         // 若第二个表未检测完，复制
}

void MergeSort(int A[], int low, int high) {
    if (low < high) {
        int mid = (low + high) / 2;    // 从中间划分
        MergeSort(A, low, mid);        // 对左半部分递归排序
        MergeSort(A, mid + 1, high);   // 对右半部分递归排序
        Merge(A, low, mid, high);      // 归并
    }
}
```

### 8. 计数排序 (Counting Sort)

**考点提示：** 牺牲空间换时间，适用于元素范围不大的场景。王道书上有提及概念。


```C++
// 假设数组A元素取值范围是 0 到 k，B是输出数组
void CountSort(int A[], int B[], int n, int k) {
    int *C = (int *)calloc(k + 1, sizeof(int)); // 申请并初始化C数组为0
    
    for (int i = 0; i < n; i++) C[A[i]]++;      // 统计每个元素出现的次数
    for (int i = 1; i <= k; i++) C[i] += C[i - 1]; // 累加，得到在B中的绝对位置
    
    // 逆序遍历A保证稳定性
    for (int i = n - 1; i >= 0; i--) {
        B[C[A[i]] - 1] = A[i];                  // 按位置填入输出数组
        C[A[i]]--;
    }
    free(C);
}
```

### 9. 基数排序 (Radix Sort)

**考点提示：** 考研卷面极少要求手写基数排序的完整链表代码，多考概念（分配、收集、LSD/MSD）。以下给出最清晰的基于数组桶的最低位优先（LSD）代码以便理解。


```C++
// 获取数组中最大数的位数
int GetMaxDigits(int A[], int n) {
    int maxVal = A[0];
    for (int i = 1; i < n; i++)
        if (A[i] > maxVal) maxVal = A[i];
    int d = 0;
    while (maxVal > 0) { maxVal /= 10; d++; }
    return d;
}

void RadixSort(int A[], int n) {
    int d = GetMaxDigits(A, n);
    int *temp = (int *)malloc(n * sizeof(int));
    int count[10]; 
    int radix = 1;
    
    for (int i = 0; i < d; i++) {             // 进行d趟分配和收集
        for (int j = 0; j < 10; j++) count[j] = 0; // 重置桶
        
        for (int j = 0; j < n; j++) {         // 统计各桶中的数量
            int k = (A[j] / radix) % 10; 
            count[k]++;
        }
        for (int j = 1; j < 10; j++) count[j] += count[j - 1]; // 前缀和累加
        
        for (int j = n - 1; j >= 0; j--) {    // 逆序收集，保证稳定
            int k = (A[j] / radix) % 10;
            temp[count[k] - 1] = A[j];
            count[k]--;
        }
        for (int j = 0; j < n; j++) A[j] = temp[j]; // 将收集好的元素复制回A
        
        radix *= 10; // 处理下一位
    }
    free(temp);
}
```
